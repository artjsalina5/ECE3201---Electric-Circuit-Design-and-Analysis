\subsection*{Formulas for Calculations (NPN Transistor)}
\begin{align*}
I_B &= \frac{V_{RB}}{R_B} \\
V_{CE} &= V_{CC} - V_{RC} - V_{RE} \\
I_C &= \frac{V_{RC}}{R_C} \\
I_E &= I_C + I_B \\
\beta &= \frac{I_C}{I_B} \\
\alpha &= \frac{I_C}{I_E}
\end{align*}
